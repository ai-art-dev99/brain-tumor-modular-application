# --- core numerics -----------------------------------------------------------
numpy>=1.26,<3
scipy>=1.11
pandas>=2.1

# --- classical ML (SVM / KNN / RF / MLP + nested CV) --------------------------
scikit-learn>=1.5
joblib>=1.3

# --- statistics: McNemar's exact test, bootstrap CIs (reviewer point 8) -------
statsmodels>=0.14

# --- CNN backbone ------------------------------------------------------------
# timm replaces the previous torch.hub NVIDIA/DeepLearningExamples call.
# Rationale: timm exposes num_classes=0 + global_pool='avg', which returns a
# true 1280-d global-average-pooled feature vector. The old code stripped the
# classifier head from the NVIDIA model, which ALSO removed the pooling layer
# and produced a 128,000-d flattened feature map.
timm>=1.0

# --- image I/O and processing ------------------------------------------------
# headless build: the container has no display server
opencv-python-headless>=4.9
pillow>=10.2
scikit-image>=0.22

# --- augmentation ------------------------------------------------------------
# albumentations replaces imgaug. imgaug is effectively unmaintained and pins
# older numpy versions, which conflicts with current scikit-learn.
albumentations>=1.4

# --- duplicate / near-duplicate detection (reviewer point 3) ------------------
imagehash>=4.3

# --- reading Figshare (Cheng) .mat files -------------------------------------
# These are MATLAB v7.3 files, i.e. HDF5 containers. scipy.io.loadmat CANNOT
# read them. h5py is required to recover cjdata.PID (patient ID), cjdata.label
# and cjdata.tumorMask. PID is what makes true patient-level splitting possible
# (reviewer point 3); tumorMask is what makes quantitative interpretability
# metrics possible (reviewer point 9).
h5py>=3.10

# --- interpretability (reviewer point 9) -------------------------------------
grad-cam>=1.5

# --- plotting ----------------------------------------------------------------

seaborn>=0.13
